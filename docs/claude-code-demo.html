from dataclasses import dataclass, field
from typing import Iterator

# Sliding-window rate limiter — O(1) amortized using a deque
from collections import deque
import time

@dataclass
class RateLimiter:
    max_calls: int
    period: float          # seconds
    _times: deque = field(default_factory=deque, repr=False)

    def allow(self) -> bool:
        now = time.monotonic()
        cutoff = now - self.period
        while self._times and self._times[0] < cutoff:
            self._times.popleft()
        if len(self._times) < self.max_calls:
            self._times.append(now)
            return True
        return False

// Generic retry with exponential back-off and jitter
async function withRetry<T>(
  fn: () => Promise<T>,
  { maxAttempts = 4, baseDelayMs = 200 } = {}
): Promise<T> {
  let lastErr: unknown;
  for (let i = 0; i < maxAttempts; i++) {
    try {
      return await fn();
    } catch (err) {
      lastErr = err;
      if (i < maxAttempts - 1) {
        const delay = baseDelayMs * 2 ** i + Math.random() * 50;
        await new Promise(r => setTimeout(r, delay));
      }
    }
  }
  throw lastErr;
}

import { useState, useCallback } from 'react';

function useLocalStorage<T>(key: string, initial: T) {
  const [value, setValue] = useState<T>(() => {
    try {
      const stored = localStorage.getItem(key);
      return stored ? JSON.parse(stored) : initial;
    } catch { return initial; }
  });

  const set = useCallback((next: T) => {
    setValue(next);
    try { localStorage.setItem(key, JSON.stringify(next)); }
    catch { /* storage quota exceeded */ }
  }, [key]);

  return [value, set] as const;
}

-- Top-N active users with their 30-day order totals
WITH recent_orders AS (
  SELECT
    user_id,
    COUNT(*) AS order_count,
    SUM(total_cents) / 100.0 AS total_usd
  FROM orders
  WHERE created_at >= NOW() - INTERVAL '30 days'
  GROUP BY user_id
)
SELECT
  u.id, u.email,
  ro.order_count,
  ro.total_usd,
  RANK() OVER (ORDER BY ro.total_usd DESC) AS rank
FROM users u
JOIN recent_orders ro ON ro.user_id = u.id
ORDER BY rank
LIMIT 10;

# Deploy: build → test → push image → restart service

log "Pushing image"
docker push "$IMAGE"

log "Restarting service"
kubectl rollout restart deployment/myapp
